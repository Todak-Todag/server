```java
@Entity class User { ... }  // DB 매핑만, 비즈니스 로직 X

// Swagger는 별도 인터페이스({Domain}ApiSpec)로 분리, Controller와 동일 패키지(presentation/controller/api 또는 controller/internal)
// Controller는 반드시 {Domain}ApiSpec을 implements하여 시그니처를 그대로 따른다. @Tag/@Operation/@ApiResponses 등 Swagger 어노테이션은 ApiSpec에만 작성하고 Controller에는 작성하지 않는다.
@Tag(name = "User", description = "사용자 API")
interface UserApiSpec {   // presentation/controller/api/, Swagger 문서화 전용 인터페이스
    @Operation(summary = "사용자 생성", description = "신규 사용자를 생성한다.")
    ApiResponse<UserCreateResponse> create(UserCreateRequest req);
}

@RestController
class UserApiController implements UserApiSpec {   // presentation/controller/api/, Validation → Service 호출 → Response 반환만
    @Override
    @PostMapping("/api/v1/users")
    public ApiResponse<UserCreateResponse> create(@Valid @RequestBody UserCreateRequest req) {
        return ApiResponse.ok("생성 성공", userCommandService.create(req.toCommand()));
    }
}

@Service
class UserCommandService {  // application/service/command/, 비즈니스 로직 + @Transactional만
    @Transactional
    UserCreateResult create(UserCreateCommand cmd) { ... }
}

// domain/repository = 추상화 인터페이스(CQRS 분리), infrastructure/persistence = JPA 구현(동일하게 분리)
// SpringData 인터페이스 자체는 나누지 않고, 이를 감싸는 Query/Command Impl만 나눈다.
interface UserQueryRepository { Optional<User> findById(UUID id); }              // domain/repository/query
interface UserCommandRepository { User save(User user); }                       // domain/repository/command
interface SpringDataUserRepository extends JpaRepository<User, UUID>, UserQueryDslRepository {} // infrastructure/persistence (분리 X)

@Repository @RequiredArgsConstructor
class UserQueryRepositoryImpl implements UserQueryRepository { // infrastructure/persistence/query
    private final SpringDataUserRepository springDataUserRepository;
    @Override public Optional<User> findById(UUID id) { return springDataUserRepository.findById(id); }
}

@Repository @RequiredArgsConstructor
class UserCommandRepositoryImpl implements UserCommandRepository { // infrastructure/persistence/command
    private final SpringDataUserRepository springDataUserRepository;
    @Override public User save(User user) { return springDataUserRepository.save(user); }
}
```

```java
// global/response/ApiResponse.java (실제 코드)
public record ApiResponse<T>(boolean success, int code, String message, T data) {
    public static <T> ApiResponse<T> of(int code, String message, T data) { return new ApiResponse<>(true, code, message, data); }
    public static <T> ApiResponse<T> ok(String message, T data) { return of(200, message, data); }
    public static <T> ApiResponse<T> created(String message, T data) { return of(201, message, data); }
}

// global/response/PageInfo.java / PageResponse.java (실제 코드)
public record PageInfo(String paginationType, int page, int size, long totalElements, int totalPages) {
    // paginationType은 항상 "OFFSET" 고정, PageInfo.of(Page<?>)로 생성
}
public record PageResponse<T>(List<T> content, PageInfo pageInfo) {
    // PageResponse.of(Page<S> page, Function<S,T> mapper)로 생성. ApiResponse<PageResponse<T>> 형태로 감싸 반환
}

// details는 고정 레코드 ErrorDetail(reason). reason은 항상 errorCode.getMessage(), message는 오버라이드 가능.
public record ErrorResponse(boolean success, String code, String message, ErrorDetail details, Instant timestamp) {
    public record ErrorDetail(String reason) {}
    public static ErrorResponse of(ErrorCode errorCode) { return of(errorCode, errorCode.getMessage()); }
    public static ErrorResponse of(ErrorCode errorCode, String message) {
        return new ErrorResponse(false, errorCode.getCode(), message, new ErrorDetail(errorCode.getMessage()), Instant.now());
    }
}

public interface ErrorCode { String getCode(); HttpStatus getStatus(); String getMessage(); }

@Getter @RequiredArgsConstructor
public enum ScheduleErrorCode implements ErrorCode {
    SERVICE_SCHEDULE_INVALID_STATUS_FOR_RESCHEDULING(HttpStatus.BAD_REQUEST, "SCHEDULED 상태의 서비스 일정만 연기할 수 있습니다."),
    SERVICE_SCHEDULE_NOT_FOUND(HttpStatus.NOT_FOUND, "존재하지 않는 서비스 일정입니다.");
    private final HttpStatus status;
    private final String message;
    @Override public String getCode() { return name(); }
}

// details Map 대신 메시지 오버라이드 방식 사용
@Getter
public class BusinessException extends RuntimeException {
    private final ErrorCode errorCode;
    public BusinessException(ErrorCode errorCode) { this(errorCode, errorCode.getMessage()); }
    public BusinessException(ErrorCode errorCode, String message) { super(message); this.errorCode = errorCode; }
}

@Slf4j @RestControllerAdvice
public class GlobalExceptionHandler {
    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<ErrorResponse> handleBusinessException(BusinessException e) {
        ErrorCode errorCode = e.getErrorCode();
        log.warn("[Schedule] 비즈니스 예외 발생 errorCode={}", errorCode.getCode());
        return ResponseEntity.status(errorCode.getStatus()).body(ErrorResponse.of(errorCode, e.getMessage()));
    }
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorResponse> handleMethodArgumentNotValid(MethodArgumentNotValidException e) {
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(ErrorResponse.of(CommonErrorCode.INVALID_PARAMETER));
    }
    @ExceptionHandler(MethodArgumentTypeMismatchException.class)
    public ResponseEntity<ErrorResponse> handleTypeMismatch(MethodArgumentTypeMismatchException e) {
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(ErrorResponse.of(CommonErrorCode.INVALID_PARAMETER));
    }
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> handleException(Exception e) {
        log.error("[Schedule] 처리되지 않은 예외 발생", e);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(ErrorResponse.of(CommonErrorCode.INTERNAL_SERVER_ERROR));
    }
}
```
